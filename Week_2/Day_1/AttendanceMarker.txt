Start
CLASS AttendanceSystem
    ATTRIBUTES: records[students][days]
    METHOD markAttendance(student, day, status)
        records[student][day] = status
    METHOD updateAttendance(student, day, newStatus)
        records[student][day] = newStatus
    METHOD monthlyReport(student)
        count = 0
        FOR each day in records[student]
            IF status == 'P' THEN count++
        RETURN count

MAIN
    CREATE AttendanceSystem with 30 days and N students
    CALL markAttendance(1, 1, 'P')
    CALL markAttendance(1, 2, 'P')
    CALL markAttendance(1, 3, 'A')
    CALL updateAttendance(1, 3, 'P')
    PRINT monthlyReport(1)
End
